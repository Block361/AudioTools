cmake_minimum_required(VERSION 3.20)
project(OpenAoIP VERSION 0.6.3 LANGUAGES CXX)
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "OpenAoIP native applications and ASIO driver require x64. Configure with -A x64.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(aoip_miniaudio STATIC src/miniaudio_impl.cpp)
target_include_directories(aoip_miniaudio SYSTEM PUBLIC third_party)
target_compile_definitions(aoip_miniaudio PUBLIC MA_NO_ENGINE MA_NO_RESOURCE_MANAGER
  MA_NO_NODE_GRAPH MA_NO_DECODING MA_NO_ENCODING MA_NO_GENERATION)
target_link_libraries(aoip_miniaudio PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_compile_definitions(aoip_miniaudio PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(aoip_miniaudio PUBLIC ole32)
elseif(APPLE)
  target_link_libraries(aoip_miniaudio PUBLIC "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
endif()
if(WIN32)
  add_library(aoip_virtual_bridge STATIC src/virtual_bridge.cpp)
  target_include_directories(aoip_virtual_bridge PUBLIC include)
  target_compile_definitions(aoip_virtual_bridge PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  add_library(aoip_vsc_engine STATIC src/vsc_engine.cpp src/wdm_client.cpp)
  target_link_libraries(aoip_vsc_engine PUBLIC aoip_session aoip_virtual_bridge)
  add_library(OpenAoIPASIO64 SHARED drivers/asio/openaoip_asio.cpp drivers/asio/openaoip_asio.def)
  target_include_directories(OpenAoIPASIO64 PRIVATE third_party/asio-sdk/ASIOSDK/common drivers/asio)
  target_link_libraries(OpenAoIPASIO64 PRIVATE aoip_virtual_bridge ole32 shell32 advapi32 uuid)
  set_target_properties(OpenAoIPASIO64 PROPERTIES PREFIX "")
  add_library(aoip_native_ui STATIC src/native/ui.cpp src/native/app_window.cpp src/native/matrix_model.cpp)
  target_include_directories(aoip_native_ui PUBLIC src/native)
  target_link_libraries(aoip_native_ui PUBLIC aoip_core user32 gdi32 shell32 ole32 windowscodecs uuid advapi32 comdlg32 dwmapi uxtheme)
  add_executable(OpenAoIPController WIN32 src/native/controller_app.cpp src/native/controller.rc)
  set_target_properties(OpenAoIPController PROPERTIES OUTPUT_NAME "OpenAoIP Controller")
  target_link_libraries(OpenAoIPController PRIVATE aoip_native_ui aoip_config)
  add_executable(OpenAoIPVirtualSoundcard WIN32 src/native/vsc_app.cpp src/native/native.rc)
  set_target_properties(OpenAoIPVirtualSoundcard PROPERTIES OUTPUT_NAME "OpenAoIP Virtual Soundcard")
  target_link_libraries(OpenAoIPVirtualSoundcard PRIVATE aoip_native_ui aoip_engine_host)
  add_executable(OpenAoIPAudioBridge WIN32 src/native/bridge_app.cpp src/native/native.rc)
  set_target_properties(OpenAoIPAudioBridge PROPERTIES OUTPUT_NAME "OpenAoIP Audio Bridge")
  target_link_libraries(OpenAoIPAudioBridge PRIVATE aoip_native_ui aoip_engine_host)
  add_executable(OpenAoIPWdmSetup WIN32 src/native/wdm_setup.cpp src/native/native.rc)
  target_link_libraries(OpenAoIPWdmSetup PRIVATE aoip_native_ui setupapi newdev wintrust crypt32)
  add_library(aoip_config src/config_store.cpp)
  target_link_libraries(aoip_config PUBLIC aoip_core shell32)
  add_library(aoip_engine_host src/engine_host.cpp)
  target_link_libraries(aoip_engine_host PUBLIC aoip_bridge_engine aoip_vsc_engine aoip_config shell32 ole32 bcrypt)
  target_compile_options(aoip_engine_host PRIVATE /utf-8 /EHsc)
  add_executable(OpenAoIPEngine src/engine_main.cpp)
  set_target_properties(OpenAoIPEngine PROPERTIES OUTPUT_NAME "OpenAoIP Engine")
  target_link_libraries(OpenAoIPEngine PRIVATE aoip_engine_host)
  foreach(desktop_target aoip_native_ui OpenAoIPController OpenAoIPVirtualSoundcard OpenAoIPAudioBridge OpenAoIPWdmSetup)
    target_compile_definitions(${desktop_target} PRIVATE UNICODE _UNICODE)
    if(MSVC)
      target_compile_options(${desktop_target} PRIVATE /utf-8 /EHsc)
    endif()
  endforeach()
  if(MSVC)
    target_link_options(OpenAoIPController PRIVATE /MANIFEST:NO)
    target_link_options(OpenAoIPVirtualSoundcard PRIVATE /MANIFEST:NO)
    target_link_options(OpenAoIPAudioBridge PRIVATE /MANIFEST:NO)
    target_link_options(OpenAoIPWdmSetup PRIVATE /MANIFEST:NO)
  endif()
endif()
if(MSVC)
  target_compile_options(aoip_miniaudio PRIVATE /w /utf-8)
endif()
add_library(aoip_audio src/audio.cpp src/audio_io.cpp src/audio_endpoint.cpp src/adaptive_buffer.cpp)
target_include_directories(aoip_audio PUBLIC include)
target_link_libraries(aoip_audio PUBLIC Threads::Threads PRIVATE aoip_miniaudio)
add_library(aoip_network src/network.cpp src/discovery.cpp)
target_include_directories(aoip_network PUBLIC include)
target_include_directories(aoip_network SYSTEM PRIVATE third_party)
target_link_libraries(aoip_network PUBLIC aoip_audio)
add_library(aoip_clock src/ptp.cpp src/ptp_client.cpp src/audio_clock.cpp)
target_include_directories(aoip_clock PUBLIC include)
target_include_directories(aoip_clock SYSTEM PUBLIC third_party)
target_link_libraries(aoip_clock PUBLIC aoip_network)
add_library(aoip_core src/controller.cpp src/devices.cpp src/receiver.cpp)
target_include_directories(aoip_core PUBLIC include)
target_include_directories(aoip_core SYSTEM PUBLIC third_party)
target_link_libraries(aoip_core PUBLIC Threads::Threads aoip_audio aoip_network)
add_library(aoip_session src/session.cpp)
target_link_libraries(aoip_session PUBLIC aoip_core aoip_clock)
add_library(aoip_bridge_engine src/bridge_engine.cpp)
target_link_libraries(aoip_bridge_engine PUBLIC aoip_session)
target_compile_definitions(aoip_core PUBLIC CPPHTTPLIB_THREAD_POOL_COUNT=4)
if(WIN32)
  target_compile_definitions(aoip_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_definitions(aoip_audio PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(aoip_network PUBLIC ws2_32 winmm iphlpapi)
endif()
if(MSVC)
  foreach(engine_target aoip_audio aoip_network aoip_clock aoip_session aoip_bridge_engine)
    target_compile_options(${engine_target} PRIVATE /W4 /utf-8 /EHsc)
  endforeach()
  target_compile_options(aoip_core PRIVATE /W4 /permissive- /utf-8 /EHsc)
else()
  target_compile_options(aoip_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(openaoip src/main.cpp)
target_link_libraries(openaoip PRIVATE aoip_core)
add_custom_target(openaoip_web ALL COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:openaoip>/web"
  DEPENDS openaoip COMMENT "Copying controller web assets")
include(CTest)

  add_executable(audio_clock_tests tests/audio_clock_tests.cpp)
  target_link_libraries(audio_clock_tests PRIVATE aoip_bridge_engine)
  add_test(NAME audio_clock_tests COMMAND audio_clock_tests)
  set_tests_properties(audio_clock_tests PROPERTIES TIMEOUT 45)
  add_executable(aoip_tests tests/tests.cpp)
  target_link_libraries(aoip_tests PRIVATE aoip_core)
  add_test(NAME aoip_tests COMMAND aoip_tests)
  set_tests_properties(aoip_tests PROPERTIES TIMEOUT 60)
  add_executable(audio_io_tests tests/audio_io_tests.cpp)
  target_link_libraries(audio_io_tests PRIVATE aoip_core)
  add_test(NAME audio_io_tests COMMAND audio_io_tests)
  set_tests_properties(audio_io_tests PROPERTIES TIMEOUT 30)
  add_executable(device_tests tests/device_tests.cpp)
  target_link_libraries(device_tests PRIVATE aoip_core)
  add_test(NAME device_tests COMMAND device_tests)
  set_tests_properties(device_tests PROPERTIES TIMEOUT 45)
  if(WIN32)
    add_executable(suite_tests tests/suite_tests.cpp)
    target_link_libraries(suite_tests PRIVATE aoip_engine_host aoip_native_ui)
    add_test(NAME suite_tests COMMAND suite_tests)
    set_tests_properties(suite_tests PROPERTIES TIMEOUT 45)
    add_executable(wdm_tests tests/wdm_tests.cpp)
    target_link_libraries(wdm_tests PRIVATE aoip_vsc_engine)
    add_test(NAME wdm_tests COMMAND wdm_tests)
    add_executable(wdm_runtime_tests tests/wdm_runtime_tests.cpp)
    target_link_libraries(wdm_runtime_tests PRIVATE aoip_vsc_engine)
    add_test(NAME wdm_runtime_tests COMMAND wdm_runtime_tests)
    set_tests_properties(wdm_runtime_tests PROPERTIES TIMEOUT 45 SKIP_RETURN_CODE 77 LABELS "wdm-runtime" RUN_SERIAL TRUE)
    add_executable(desktop_tests tests/desktop_tests.cpp)
    target_link_libraries(desktop_tests PRIVATE aoip_native_ui aoip_vsc_engine)
    add_test(NAME desktop_tests COMMAND desktop_tests)
    set_tests_properties(desktop_tests PROPERTIES TIMEOUT 30)
    add_executable(asio_bridge_tests drivers/asio/test_host.cpp)
    target_include_directories(asio_bridge_tests PRIVATE drivers/asio third_party/asio-sdk/ASIOSDK/common)
    target_link_libraries(asio_bridge_tests PRIVATE aoip_vsc_engine ole32 uuid)
    add_test(NAME asio_bridge_tests COMMAND asio_bridge_tests "$<TARGET_FILE:OpenAoIPASIO64>")
    set_tests_properties(asio_bridge_tests PROPERTIES TIMEOUT 40)
  endif()
  if(WIN32)
    find_program(POWERSHELL_EXECUTABLE powershell)
    if(POWERSHELL_EXECUTABLE)
      add_test(NAME vm_workflow_smoke COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/vm-workflow-smoke.ps1" -SourceDirectory "${CMAKE_CURRENT_SOURCE_DIR}")
      set_tests_properties(vm_workflow_smoke PROPERTIES TIMEOUT 20)
      add_test(NAME wdm_installer_smoke COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/wdm-installer-smoke.ps1" -BinaryDirectory "$<TARGET_FILE_DIR:OpenAoIPController>" -Inf "${CMAKE_CURRENT_SOURCE_DIR}/drivers/wdm/OpenAoIPWdm.inf")
      set_tests_properties(wdm_installer_smoke PROPERTIES TIMEOUT 20)
      add_test(NAME native_smoke COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/native-smoke.ps1" -BinaryDirectory "$<TARGET_FILE_DIR:OpenAoIPController>")
      set_tests_properties(native_smoke PROPERTIES TIMEOUT 65 RUN_SERIAL TRUE)
      add_test(NAME package_smoke COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/package-smoke.ps1" -BinaryDirectory "$<TARGET_FILE_DIR:OpenAoIPController>")
      set_tests_properties(package_smoke PROPERTIES TIMEOUT 40)
      add_test(NAME process_smoke COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile
        -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/process-smoke.ps1"

  endif()

install(TARGETS openaoip RUNTIME DESTINATION . COMPONENT Tools)
install(DIRECTORY web examples DESTINATION . COMPONENT Tools)
install(FILES README.md NETZWERK-START.md WINDOWS-START.md WINDOWS-AUDIO.md ARCHITECTURE.md PTP.md LICENSE THIRD_PARTY.md DESTINATION . COMPONENT Tools)
install(FILES third_party/LICENSE-httplib.txt third_party/LICENSE-json.txt third_party/LICENSE-miniaudio.txt DESTINATION third_party COMPONENT Tools)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_NAME OpenAoIP)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "OpenAoIP-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
set(CPACK_ARCHIVE_COMPONENT_INSTALL ON)
set(CPACK_COMPONENTS_ALL Controller VirtualSoundcard AudioBridge Suite Tools)
set(CPACK_ARCHIVE_CONTROLLER_FILE_NAME "OpenAoIP-Controller-${PROJECT_VERSION}-Windows-x64")
set(CPACK_ARCHIVE_VIRTUALSOUNDCARD_FILE_NAME "OpenAoIP-VirtualSoundcard-${PROJECT_VERSION}-Windows-x64")
set(CPACK_ARCHIVE_AUDIOBRIDGE_FILE_NAME "OpenAoIP-AudioBridge-${PROJECT_VERSION}-Windows-x64")
set(CPACK_ARCHIVE_SUITE_FILE_NAME "OpenAoIP-Suite-${PROJECT_VERSION}-Windows-x64")
set(CPACK_ARCHIVE_TOOLS_FILE_NAME "OpenAoIP-Tools-${PROJECT_VERSION}-Windows-x64")

include(cmake/installer.cmake)
